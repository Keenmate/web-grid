SP05 · API

// Set initial sort
grid.sort = [
    { column: 'dept', direction: 'asc' },
    { column: 'name', direction: 'desc' }
]

// Page size selector
grid.pageSizes = [10, 25, 50, 100]
grid.pageSize = 25  // Current page size

// Pagination position
// Values: top-left, top-center, top-right,
//         bottom-left, bottom-center, bottom-right
grid.paginationPosition = 'bottom-right'
// Multiple positions with pipe separator:
grid.paginationPosition = 'top-right|bottom-right'

// Server-side pagination
grid.totalItems = 500
grid.currentPage = 1
grid.isPaginationShown = true

// Handle data requests
grid.onDataRequest = (detail) => {
    // detail: { sort, page, pageSize, trigger }
    // trigger: 'sort' | 'page' | 'page-size' | 'init'
    fetchData(detail).then(res => {
        grid.items = res.data
        grid.totalItems = res.total
    })
}

// Custom/translated labels
grid.getPaginationLabelsCallback = (ctx) => ({
    previous: '← Back',
    next: 'Forward →',
    pageInfo: `${ctx.currentPage}/${ctx.totalPages}`,
    itemCount: `${ctx.totalItems} records`,
    perPage: 'rows'
})

// Summary (aggregates, totals)
grid.summaryPosition = 'bottom-left'
grid.renderSummaryCallback = ({ allItems }) => {
  const total = allItems.reduce((s, r) => s + r.salary, 0)
  return `Total: $${total.toLocaleString()}`
}
